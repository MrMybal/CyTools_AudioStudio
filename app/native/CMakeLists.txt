cmake_minimum_required(VERSION 3.22)
project(AudioStudioPluginHost VERSION 1.0.0 LANGUAGES C CXX)
include(FetchContent)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# Offline builds may supply -DFETCHCONTENT_SOURCE_DIR_JUCE=/path/to/JUCE.
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 91ad83ae34a81e0833b1a2b0866f54846370ae53)
FetchContent_MakeAvailable(JUCE)
set(YSFX_PLUGIN OFF CACHE BOOL "" FORCE)
set(YSFX_TESTS OFF CACHE BOOL "" FORCE)
set(YSFX_SKIP_CHECKSUM ON CACHE BOOL "" FORCE)
find_program(NASM_PROGRAM nasm)
if(MSVC AND NOT NASM_PROGRAM)
    set(YSFX_PORTABLE ON CACHE BOOL "" FORCE)
endif()
set(CMAKE_BUILD_TYPE "${CMAKE_BUILD_TYPE}" CACHE STRING "Build type")
add_subdirectory(third_party/ysfx EXCLUDE_FROM_ALL)
juce_add_console_app(AudioStudioPluginWorker PRODUCT_NAME "AudioStudio Plugin Worker")
target_sources(AudioStudioPluginWorker PRIVATE PluginWorkerMain.cpp ExtraPluginFormats.cpp ClapInstance.cpp JsfxInstance.cpp)
target_compile_features(AudioStudioPluginWorker PRIVATE cxx_std_20)
target_include_directories(AudioStudioPluginWorker PRIVATE include third_party/clap/include)
target_compile_definitions(AudioStudioPluginWorker PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN
    JUCE_PLUGINHOST_VST=0 JUCE_PLUGINHOST_VST3=0 JUCE_PLUGINHOST_LV2=1
    JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(AudioStudioPluginWorker PRIVATE ysfx::ysfx juce::juce_audio_processors juce::juce_gui_extra juce::juce_recommended_config_flags)
install(TARGETS AudioStudioPluginWorker RUNTIME DESTINATION . COMPONENT AudioStudioHost)
